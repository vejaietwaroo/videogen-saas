async function triggerRender() { const promptText = document.getElementById("renderPrompt").value.trim(); const aspectRatio = document.getElementById("videoFormat").value; const avatar = document.getElementById("avatarSelect").value; const voicePersona = document.getElementById("voicePersona").value; const dialogueText = document.getElementById("dialogueScript").value.trim(); const musicTrack = document.getElementById("musicTrack").value; const autoCaptions = document.getElementById("autoCaptions").checked; const renderBtn = document.getElementById("renderBtn"); const renderError = document.getElementById("renderError"); renderError.style.display = "none"; if (!authToken) { renderError.innerText = "Please activate your license key first."; renderError.style.display = "block"; return; } if (!promptText) { renderError.innerText = "Please describe the visual scene you want to generate."; renderError.style.display = "block"; return; } const previewCard = document.getElementById("previewCard"); const renderSpinner = document.getElementById("renderSpinner"); const renderStatusText = document.getElementById("renderStatusText"); const previewIcon = document.getElementById("previewIcon"); const previewTitle = document.getElementById("previewTitle"); const previewSubtitle = document.getElementById("previewSubtitle"); const downloadBtn = document.getElementById("downloadBtn"); const videoContainer = document.getElementById("videoContainer"); const placeholderBox = document.getElementById("placeholderBox"); const resultVideo = document.getElementById("resultVideo"); previewCard.style.display = "block"; renderSpinner.style.display = "block"; renderStatusText.style.color = "#f59e0b"; renderStatusText.innerText = "Your video is rendering..."; videoContainer.style.display = "none"; resultVideo.src = ""; placeholderBox.style.display = "block"; previewIcon.innerText = "🎬"; previewTitle.innerText = "Generating Your Video"; previewSubtitle.innerText = `Preparing ${aspect_ratio_label(aspectRatio)} render on cloud node...`; document.getElementById("renderedPromptDisplay").innerText = `"${promptText}"`; downloadBtn.style.display = "none"; renderBtn.disabled = true; renderBtn.innerText = "Rendering..."; try { const response = await fetch(`${API_BASE}/api/render`, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${authToken}` }, body: JSON.stringify({ prompt: promptText, aspect_ratio: aspectRatio, avatar: avatar, voice: voicePersona, dialogue: dialogueText, music: musicTrack, captions: autoCaptions }) }); const data = await response.json(); if (response.ok) { renderSpinner.style.display = "none"; renderStatusText.style.color = "#34d399"; renderStatusText.innerText = "✓ Render Complete"; if (data.video_url) { placeholderBox.style.display = "none"; resultVideo.src = data.video_url; videoContainer.style.display = "block"; downloadBtn.href = data.video_url; downloadBtn.style.display = "block"; addToHistory(promptText, aspectRatio, data.video_url); } } else { previewCard.style.display = "none"; renderError.innerText = data.detail || "Unable to complete render request."; renderError.style.display = "block"; } } catch (err) { previewCard.style.display = "none"; renderError.innerText = "Connection interrupted while queuing your video."; renderError.style.display = "block"; } finally { renderBtn.disabled = false; renderBtn.innerText = "Generate Video"; } }
